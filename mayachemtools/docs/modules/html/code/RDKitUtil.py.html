   92 def GetMolName(Mol, MolNum=None):
   93     """Get molecule name.
   94 
   95     Arguments:
   96         Mol (object): RDKit molecule object.
   97         MolNum (int or None): Molecule number in input file.
   98 
   99     Returns:
  100         str : Molname corresponding to _Name property of a molecule, generated
  101             from specieid MolNum using the format "Mol%d" % MolNum, or an
  102             empty string.
  103 
  104     """
  105 
  106     MolName = ""
  107     if Mol.HasProp("_Name"):
  108         MolName = Mol.GetProp("_Name")
  109 
  110     if not len(MolName):
  111         if MolNum is not None:
  112             MolName = "Mol%d" % MolNum
  113 
  114     return MolName

  189 def _ModifySVGForInlineEmbedding(SVGText, Base64Encoded):
  190     """Modify SVG for inline embedding into a HTML page using "img" tag
  191     along with performing base64 encoding.
  192     """
  193 
  194     # Take out all tags till the start of '<svg' tag...
  195     Pattern = re.compile("^.*<svg", re.I | re.S)
  196     SVGText = Pattern.sub("<svg", SVGText)
  197 
  198     # Add an extra space before the "width=..." tag. Otherwise, inline embedding may
  199     # cause the following XML error on some browsers due to start of the "width=..."
  200     # at the begining of the line in <svg ...> tag:
  201     #
  202     #  XML5607: Whitespace expected.
  203     #
  204     SVGText = re.sub("width='", " width='", SVGText, flags=re.I)
  205 
  206     # Take out trailing new line...
  207     SVGText = SVGText.strip()
  208 
  209     # Perform base64 encoding by turning text into byte stream using string
  210     # encode and transform byte stream returned by b64encode into a string
  211     # by string decode...
  212     #
  213     if Base64Encoded:
  214         SVGText = base64.b64encode(SVGText.encode()).decode()
  215 
  216     return SVGText

  295 def _ModifySVGForBrowserViewing(SVGText, BoldText=True):
  296     """Modify SVG for loading into a browser."""
  297 
  298     # It appears that the string 'xmlns:svg' needs to be replaced with 'xmlns' in the
  299     # SVG image string generated by older versions of RDKit. Otherwise, the image
  300     # doesn't load in web browsers.
  301     #
  302     if re.search("xmlns:svg", SVGText, re.I):
  303         SVGText = re.sub("xmlns:svg", "xmlns", SVGText, flags=re.I)
  304 
  305     # Make text bold...
  306     if BoldText:
  307         SVGText = re.sub("font-weight:normal;", "font-weight:bold;", SVGText, flags=re.I)
  308 
  309     return SVGText

  328 def IsAtomSymbolPresentInMol(Mol, AtomSymbol, IgnoreCase=True):
  329     """Check for the presence of an atom symbol in a molecule.
  330 
  331     Arguments:
  332         Mol (object): RDKit molecule object.
  333         AtomSymbol (str): Atom symbol.
  334 
  335     Returns:
  336         bool : True - Atom symbol in molecule; Otherwise, false.
  337 
  338     """
  339 
  340     for Atom in Mol.GetAtoms():
  341         Symbol = Atom.GetSymbol()
  342         if IgnoreCase:
  343             if re.match("^%s$" % AtomSymbol, Symbol, re.I):
  344                 return True
  345         else:
  346             if re.match("^%s$" % AtomSymbol, Symbol):
  347                 return True
  348 
  349     return False

  439 def GetFormalCharge(Mol, CheckMolProp=True):
  440     """Get formal charge of a molecule. The formal charge is either retrieved
  441     from 'FormalCharge' molecule property or calculated using RDKit function
  442     Chem.GetFormalCharge(Mol).
  443 
  444     The 'FormalCharge' molecule property may contain multiple space delimited
  445     values. The total formal charge corresponds to the sum of the specified formal
  446     charge values.
  447 
  448     Arguments:
  449         Mol (object): RDKit molecule object.
  450         CheckMolProp (bool): Check 'FormalCharge' molecule property to
  451             retrieve formal charge.
  452 
  453     Returns:
  454         int : Formal charge.
  455 
  456     """
  457 
  458     Name = "FormalCharge"
  459     if CheckMolProp and Mol.HasProp(Name):
  460         FormalCharge = Mol.GetProp(Name)
  461         Values = FormalCharge.split()
  462         if len(Values) > 1:
  463             MiscUtil.PrintWarning(
  464                 "RDKitUtil.GetFormalCharge: Molecule property, %s, contains multiple values, %s. Formal charge corresponds to sum of the specified values..."
  465                 % (Name, FormalCharge)
  466             )
  467             FormalCharge = 0.0
  468             for Value in Values:
  469                 FormalCharge += float(Value)
  470             FormalCharge = int(FormalCharge)
  471         else:
  472             FormalCharge = int(float(FormalCharge))
  473     else:
  474         FormalCharge = CalculateFormalCharge(Mol)
  475 
  476     return int(FormalCharge)

  495 def GetSpinMultiplicity(Mol, CheckMolProp=True):
  496     """Get spin multiplicity of a molecule. The spin multiplicity is either
  497     retrieved from 'SpinMultiplicity' molecule property or calculated
  498     from the number of free radical electrons using Hund's rule of maximum
  499     multiplicity defined as 2S + 1 where S is the total electron spin. The
  500     total spin is 1/2 the number of free radical electrons in a molecule.
  501 
  502     The 'SpinMultiplicity' molecule property may contain multiple space delimited
  503     values. The total spin multiplicity corresponds to the total number of free radical
  504     electrons which are calculated for each specified value.
  505 
  506     Arguments:
  507         Mol (object): RDKit molecule object.
  508         CheckMolProp (bool): Check 'SpinMultiplicity' molecule property to
  509             retrieve spin multiplicity.
  510 
  511     Returns:
  512         int : Spin multiplicity.
  513 
  514     """
  515 
  516     Name = "SpinMultiplicity"
  517     if CheckMolProp and Mol.HasProp(Name):
  518         SpinMultiplicity = Mol.GetProp(Name)
  519         Values = SpinMultiplicity.split()
  520         if len(Values) > 1:
  521             MiscUtil.PrintWarning(
  522                 "RDKitUtil.GetSpinMultiplicity: Molecule property, %s, contains multiple values, %s. Calculating spin multiplicity corresponding to total number of free radical electrons for each specified value..."
  523                 % (Name, SpinMultiplicity)
  524             )
  525             NumRadicalElectrons = 0
  526             for Value in Values:
  527                 NumRadicalElectrons += int(float(Value)) - 1
  528 
  529             TotalElectronicSpin = NumRadicalElectrons / 2
  530             SpinMultiplicity = 2 * TotalElectronicSpin + 1
  531         else:
  532             SpinMultiplicity = int(float(SpinMultiplicity))
  533     else:
  534         SpinMultiplicity = CalculateSpinMultiplicity(Mol)
  535 
  536     return int(SpinMultiplicity)

  539 def CalculateSpinMultiplicity(Mol):
  540     """Calculate spin multiplicity of a molecule. The spin multiplicity is calculated
  541     from the number of free radical electrons using Hund's rule of maximum
  542     multiplicity defined as 2S + 1 where S is the total electron spin. The
  543     total spin is 1/2 the number of free radical electrons in a molecule.
  544 
  545     Arguments:
  546         Mol (object): RDKit molecule object.
  547 
  548     Returns:
  549         int : Spin multiplicity.
  550 
  551     """
  552 
  553     # Calculate spin multiplicity using Hund's rule of maximum multiplicity...
  554     NumRadicalElectrons = 0
  555     for Atom in Mol.GetAtoms():
  556         NumRadicalElectrons += Atom.GetNumRadicalElectrons()
  557 
  558     TotalElectronicSpin = NumRadicalElectrons / 2
  559     SpinMultiplicity = 2 * TotalElectronicSpin + 1
  560 
  561     return int(SpinMultiplicity)

  564 def GetPsi4XYZFormatString(
  565     Mol,
  566     ConfID=-1,
  567     FormalCharge="auto",
  568     SpinMultiplicity="auto",
  569     Symmetry="auto",
  570     NoCom=False,
  571     NoReorient=False,
  572     CheckFragments=False,
  573 ):
  574     """Retrieve geometry string of a molecule in Psi4ish XYZ format to perform
  575     Psi4 quantum chemistry calculations.
  576 
  577     You may explicit specify multiple space delimited values for formal charge
  578     and spin multiplicity. Otherwise, these values are either automatically
  579     retrieved from 'FormalCharge' and 'SpinMultiplicity' molecule properties or
  580     calculated using RDKit. The number of specified values for these properties
  581     must match the number of fragments in the molecule during the processing
  582     of the fragments.
  583 
  584     Arguments:
  585         Mol (object): RDKit molecule object.
  586         ConfID (int): Conformer number.
  587         FormalCharge (str): Specified formal charge or 'auto' to calculate
  588            its value by RDKit.
  589         SpinMultiplicity (str): Specified spin multiplicity or 'auto' to calculate
  590            its value by RDKit.
  591         Symmetry (str): Specified symmetry or 'auto' to calculate its value by
  592            Psi4.
  593         NoCom (bool): Flag to disable recentering of a molecule by Psi4.
  594         NoReorient (bool): Flag to disable reorientation of a molecule by Psi4.
  595         CheckFragments (bool): Check for fragments and setup geometry string
  596            using  -- separator between fragments.
  597 
  598     Returns:
  599         str : Geometry string of a molecule in Psi4ish XYZ format.
  600 
  601     """
  602 
  603     # Check for fragments...
  604     Mols = [Mol]
  605     if CheckFragments:
  606         Fragments = list(Chem.rdmolops.GetMolFrags(Mol, asMols=True))
  607         if len(Fragments) > 1:
  608             Mols = Fragments
  609 
  610     FragMolFormalCharges = _SetFormalChargesForPsi4XYZFormatString(Mol, Mols, FormalCharge, CheckFragments)
  611     FragMolSpinMultiplicities = _SetSpinMultiplicitiesForPsi4XYZFormatString(
  612         Mol, Mols, SpinMultiplicity, CheckFragments
  613     )
  614 
  615     # Setup geometry string for Ps4...
  616     GeometryList = []
  617     FragMolCount = 0
  618 
  619     for FragMolIndex, FragMol in enumerate(Mols):
  620         FragMolCount += 1
  621         if FragMolCount > 1:
  622             GeometryList.append("--")
  623 
  624         FragMolFormalCharge = FragMolFormalCharges[FragMolIndex]
  625         FragMolSpinMultiplicity = FragMolSpinMultiplicities[FragMolIndex]
  626         if FragMolFormalCharge is None or FragMolSpinMultiplicity is None:
  627             MiscUtil.PrintInfo("")
  628             MiscUtil.PrintWarning(
  629                 "RDKitUtil.GetPsi4XYZFormatString: Failed to set formal charge and spin multiplicity values. Both formal charge, %s, and spin multiplicity, %s, must be valid values. These values are either specified explicitly or automatically calculated..."
  630                 % (FragMolFormalCharge, FragMolSpinMultiplicity)
  631             )
  632         else:
  633             GeometryList.append("%s %s" % (FragMolFormalCharge, FragMolSpinMultiplicity))
  634 
  635         AtomSymbols = GetAtomSymbols(FragMol)
  636         AtomPositions = GetAtomPositions(FragMol, ConfID)
  637 
  638         for AtomSymbol, AtomPosition in zip(AtomSymbols, AtomPositions):
  639             GeometryList.append("%s %s %s %s" % (AtomSymbol, AtomPosition[0], AtomPosition[1], AtomPosition[2]))
  640 
  641     GeometryList.append("units angstrom")
  642 
  643     if not re.match("^auto$", Symmetry, re.I):
  644         Name = "Symmetry"
  645         if Mol.HasProp(Name):
  646             Symmetry = Mol.GetProp(Name)
  647         GeometryList.append("symmetry %s" % Symmetry)
  648 
  649     if NoCom:
  650         GeometryList.append("no_com")
  651 
  652     if NoReorient:
  653         GeometryList.append("no_reorient")
  654 
  655     Geometry = "\n".join(GeometryList)
  656 
  657     return Geometry

  660 def _SetFormalChargesForPsi4XYZFormatString(Mol, FragMols, FormalCharge, CheckFragments):
  661     """Setup formal charges for Psi4 XYZ format string."""

  942 def ReorderAtomIndicesInSequentiallyConnectedManner(Mol, AtomIndices):
  943     """Check for the presence of sequentially connected list of atoms in an
  944     arbitray list of atoms in molecule.
  945 
  946     Arguments:
  947         Mol (object): RDKit molecule object.
  948         AtomIndices (list): List of atom indices.
  949 
  950     Returns:
  951         bool : True - Sequentially connected list found; Otherwise, false.
  952         list : List of seqeuntially connected atoms or None.
  953 
  954     """
  955 
  956     # Count the number of neighbors for specified atom indices ensuring
  957     # that the neighbors are also part of atom indices...
  958     AtomNbrsCount = {}
  959     for AtomIndex in AtomIndices:
  960         Atom = Mol.GetAtomWithIdx(AtomIndex)
  961 
  962         AtomNbrsCount[AtomIndex] = 0
  963         for AtomNbr in Atom.GetNeighbors():
  964             AtomNbrIndex = AtomNbr.GetIdx()
  965             if AtomNbrIndex not in AtomIndices:
  966                 continue
  967             AtomNbrsCount[AtomIndex] += 1
  968 
  969     # Number of neighbors for each specified atom indices must be 1 or 2
  970     # for sequentially connected list of atom indices...
  971     AtomsWithOneNbr = []
  972     for AtomIndex, NbrsCount in AtomNbrsCount.items():
  973         if not (NbrsCount == 1 or NbrsCount == 2):
  974             return (False, None)
  975 
  976         if NbrsCount == 1:
  977             AtomsWithOneNbr.append(AtomIndex)
  978 
  979     # A sequentially connected list of indices must have two atom indices with
  980     # exactly # one neighbor...
  981     if len(AtomsWithOneNbr) != 2:
  982         return (False, None)
  983 
  984     # Setup a reordered list of sequentially connected atoms...
  985     ReorderedAtomIndices = []
  986 
  987     AtomIndex1, AtomIndex2 = AtomsWithOneNbr
  988     AtomIndex = AtomIndex1 if AtomIndex1 < AtomIndex2 else AtomIndex2
  989     ReorderedAtomIndices.append(AtomIndex)
  990 
  991     while len(ReorderedAtomIndices) < len(AtomIndices):
  992         Atom = Mol.GetAtomWithIdx(AtomIndex)
  993 
  994         for AtomNbr in Atom.GetNeighbors():
  995             AtomNbrIndex = AtomNbr.GetIdx()
  996             if AtomNbrIndex not in AtomIndices:
  997                 continue
  998 
  999             if AtomNbrIndex in ReorderedAtomIndices:
 1000                 continue
 1001 
 1002             # Treat neighbor as next connected atom...
 1003             AtomIndex = AtomNbrIndex
 1004             ReorderedAtomIndices.append(AtomIndex)
 1005             break
 1006 
 1007     # Check reorderd list size...
 1008     if len(ReorderedAtomIndices) != len(AtomIndices):
 1009         return (False, None)
 1010 
 1011     # A final check to validate reorderd list...
 1012     if not AreAtomIndicesSequentiallyConnected(Mol, ReorderedAtomIndices):
 1013         return (False, None)
 1014 
 1015     return (True, ReorderedAtomIndices)

 1473 def ConstrainAndEmbed(
 1474     mol,
 1475     core,
 1476     coreMatchesMol=None,
 1477     useTethers=True,
 1478     coreConfId=-1,
 1479     randomseed=2342,
 1480     getForceField=AllChem.UFFGetMoleculeForceField,
 1481     **kwargs,
 1482 ):
 1483     """
 1484     The function is a local copy of RDKit fucntion AllChem.ConstrainedEmbed().
 1485     It has been enhanced to support an explicit list of core matches corresponding
 1486     to the matched atom indices in the molecule. The number of matched atom indices
 1487     must be equal to the number of atoms in core molecule.
 1488 
 1489     Arguments:
 1490         mol (object): RDKit molecule object to embed.
 1491         core (object): RDKit molecule to use as a source of constraints.
 1492         coreMatchesMol (list): A list matches atom indices in mol.
 1493         useTethers: (bool) if True, the final conformation will be optimized
 1494             subject to a series of extra forces that pull the matching atoms to
 1495             the positions of the core atoms. Otherwise simple distance
 1496             constraints based on the core atoms will be used in the
 1497             optimization.
 1498         coreConfId (int): ID of the core conformation to use.
 1499         randomSeed (int): Seed for the random number generator
 1500 
 1501     Returns:
 1502         mol (object): RDKit molecule object.
 1503 
 1504     """
 1505     if coreMatchesMol is None:
 1506         match = mol.GetSubstructMatch(core)
 1507         if not match:
 1508             raise ValueError("Molecule doesn't match the core.")
 1509     else:
 1510         if core.GetNumAtoms() != len(coreMatchesMol):
 1511             raise ValueError(
 1512                 "Number of atoms, %s, in core molecule must match number of atom indices, %s, specified in the list coreMatchesMol."
 1513                 % (core.GetNumAtoms(), len(coreMatchesMol))
 1514             )
 1515         # Check specified matched atom indices in  coreMatchesMol and use the match atom
 1516         # indices returned by GetSubstructMatches() for embedding...
 1517         coreMatch = None
 1518         matches = mol.GetSubstructMatches(core)
 1519         for match in matches:
 1520             if len(match) != len(coreMatchesMol):
 1521                 continue
 1522             matchFound = True
 1523             for atomIndex in match:
 1524                 if atomIndex not in coreMatchesMol:
 1525                     matchFound = False
 1526                     break
 1527             if matchFound:
 1528                 coreMatch = match
 1529                 break
 1530         if coreMatch is None:
 1531             raise ValueError("Molecule doesn't match the atom indices specified in the list coreMatchesMol.")
 1532         match = coreMatch
 1533 
 1534     coordMap = {}
 1535     coreConf = core.GetConformer(coreConfId)
 1536     for i, idxI in enumerate(match):
 1537         corePtI = coreConf.GetAtomPosition(i)
 1538         coordMap[idxI] = corePtI
 1539 
 1540     ci = AllChem.EmbedMolecule(mol, coordMap=coordMap, randomSeed=randomseed, **kwargs)
 1541     if ci < 0:
 1542         raise ValueError("Could not embed molecule.")
 1543 
 1544     algMap = [(j, i) for i, j in enumerate(match)]
 1545 
 1546     if not useTethers:
 1547         # clean up the conformation
 1548         ff = getForceField(mol, confId=0)
 1549         for i, idxI in enumerate(match):
 1550             for j in range(i + 1, len(match)):
 1551                 idxJ = match[j]
 1552                 d = coordMap[idxI].Distance(coordMap[idxJ])
 1553                 ff.AddDistanceConstraint(idxI, idxJ, d, d, 100.0)
 1554         ff.Initialize()
 1555         n = 4
 1556         more = ff.Minimize()
 1557         while more and n:
 1558             more = ff.Minimize()
 1559             n -= 1
 1560         # rotate the embedded conformation onto the core:
 1561         rms = AllChem.AlignMol(mol, core, atomMap=algMap)
 1562     else:
 1563         # rotate the embedded conformation onto the core:
 1564         rms = AllChem.AlignMol(mol, core, atomMap=algMap)
 1565         ff = getForceField(mol, confId=0)
 1566         conf = core.GetConformer()
 1567         for i in range(core.GetNumAtoms()):
 1568             p = conf.GetAtomPosition(i)
 1569             pIdx = ff.AddExtraPoint(p.x, p.y, p.z, fixed=True) - 1
 1570             ff.AddDistanceConstraint(pIdx, match[i], 0, 0, 100.0)
 1571         ff.Initialize()
 1572         n = 4
 1573         more = ff.Minimize(energyTol=1e-4, forceTol=1e-3)
 1574         while more and n:
 1575             more = ff.Minimize(energyTol=1e-4, forceTol=1e-3)
 1576             n -= 1
 1577         # realign
 1578         rms = AllChem.AlignMol(mol, core, atomMap=algMap)
 1579 
 1580     mol.SetProp("EmbedRMS", str(rms))
 1581     return mol
