  130 def ProcessMoleculesUsingSingleProcess(Mols):
  131     """Process molecules to perform torsion scan using a single process."""
  132 
  133     MolInfoText = "first molecule"
  134     if not OptionsInfo["FirstMolMode"]:
  135         MolInfoText = "all molecules"
  136 
  137     if OptionsInfo["TorsionMinimize"]:
  138         MiscUtil.PrintInfo(
  139             "\nPerforming torsion scan on %s by generating conformation ensembles for specific torsion angles and constrained energy minimization of the ensembles..."
  140             % (MolInfoText)
  141         )
  142     else:
  143         MiscUtil.PrintInfo(
  144             "\nPerforming torsion scan on %s by skipping generation of conformation ensembles for specific torsion angles and constrained energy minimization of the ensembles..."
  145             % (MolInfoText)
  146         )
  147 
  148     SetupTorsionsPatternsInfo()
  149 
  150     (MolCount, ValidMolCount, TorsionsMissingCount, MinimizationFailedCount, TorsionsScanFailedCount) = [0] * 5
  151 
  152     for Mol in Mols:
  153         MolCount += 1
  154 
  155         if OptionsInfo["FirstMolMode"] and MolCount > 1:
  156             MolCount -= 1
  157             break
  158 
  159         if Mol is None:
  160             continue
  161 
  162         if RDKitUtil.IsMolEmpty(Mol):
  163             if not OptionsInfo["QuietMode"]:
  164                 MolName = RDKitUtil.GetMolName(Mol, MolCount)
  165                 MiscUtil.PrintWarning("Ignoring empty molecule: %s" % MolName)
  166             continue
  167         ValidMolCount += 1
  168 
  169         Mol, MinimizationCalcStatus, TorsionsMatchStatus, TorsionsScanCalcStatus = PerformMinimizationAndTorsionScan(
  170             Mol, MolCount
  171         )
  172 
  173         if not MinimizationCalcStatus:
  174             MinimizationFailedCount += 1
  175             continue
  176 
  177         if not TorsionsMatchStatus:
  178             TorsionsMissingCount += 1
  179             continue
  180 
  181         if not TorsionsScanCalcStatus:
  182             TorsionsScanFailedCount += 1
  183             continue
  184 
  185     return (MolCount, ValidMolCount, MinimizationFailedCount, TorsionsMissingCount, TorsionsScanFailedCount)

  280             TorsionsScanFailedCount += 1

  307 
  308     (MinimizationCalcStatus, TorsionsMatchStatus, TorsionsScanCalcStatus) = [False] * 3
  309

  311         return [MolIndex, None, MinimizationCalcStatus, TorsionsMatchStatus, TorsionsScanCalcStatus]

  314     if RDKitUtil.IsMolEmpty(Mol):

  317             MiscUtil.PrintWarning("Ignoring empty molecule: %s" % MolName)

  320     Mol, MinimizationCalcStatus, TorsionsMatchStatus, TorsionsScanCalcStatus = PerformMinimizationAndTorsionScan(
  321         Mol, (MolIndex + 1)
  322     )
  323 
  324     return [
  325         MolIndex,
  326         RDKitUtil.MolToBase64EncodedMolString(
  327             Mol, PropertyPickleFlags=Chem.PropertyPickleOptions.MolProps | Chem.PropertyPickleOptions.PrivateProps

  393 def ScanSingleTorsionInMolUsingMultipleProcessesAtTorsionAnglesLevel(
  394     Mol, TorsionID, TorsionPattern, TorsionPatternMol, TorsionMatches, MolNum
  395 ):
  396     """Perform torsion scan for a molecule using multiple processses at torsion angles
  397     level along with constrained energy minimization.
  398     """
  399 
  400     if OptionsInfo["MPLevelMoleculesMode"]:
  401         MiscUtil.PrintError(
  402             "Single torison scanning for a molecule is not allowed in multiprocessing mode at molecules level.\n"
  403         )
  404 
  405     Mols, Angles = SetupMolsForSingleTorsionScanInMol(Mol, TorsionMatches, MolNum)
  406 
  407     MPParams = OptionsInfo["MPParams"]
  408 
  409     # Setup data for initializing a worker process...
  410     MiscUtil.PrintInfo("\nEncoding options info...")
  411 
  412     # Track and avoid encoding TorsionsPatternsInfo as it contains RDKit molecule object...
  413     TorsionsPatternsInfo = OptionsInfo["TorsionsPatternsInfo"]
  414     OptionsInfo["TorsionsPatternsInfo"] = None
  415 
  416     InitializeWorkerProcessArgs = (
  417         MiscUtil.ObjectToBase64EncodedString(Options),
  418         MiscUtil.ObjectToBase64EncodedString(OptionsInfo),
  419     )
  420 
  421     # Restore TorsionsPatternsInfo...
  422     OptionsInfo["TorsionsPatternsInfo"] = TorsionsPatternsInfo
  423 
  424     # Setup a encoded mols data iterable for a worker process...
  425     WorkerProcessDataIterable = GenerateBase64EncodedMolStringsWithTorsionScanInfo(
  426         Mol, (MolNum - 1), Mols, Angles, TorsionID, TorsionPattern, TorsionPatternMol, TorsionMatches
  427     )
  428 
  429     # Setup process pool along with data initialization for each process...
  430     MiscUtil.PrintInfo(
  431         "\nConfiguring multiprocessing using %s method..."
  432         % ("mp.Pool.imap()" if re.match("^Lazy$", MPParams["InputDataMode"], re.I) else "mp.Pool.map()")
  433     )
  434     MiscUtil.PrintInfo(
  435         "NumProcesses: %s; InputDataMode: %s; ChunkSize: %s\n"
  436         % (
  437             MPParams["NumProcesses"],
  438             MPParams["InputDataMode"],
  439             ("automatic" if MPParams["ChunkSize"] is None else MPParams["ChunkSize"]),
  440         )
  441     )
  442 
  443     ProcessPool = mp.Pool(MPParams["NumProcesses"], InitializeTorsionAngleWorkerProcess, InitializeWorkerProcessArgs)
  444 
  445     # Start processing...
  446     if re.match("^Lazy$", MPParams["InputDataMode"], re.I):
  447         Results = ProcessPool.imap(TorsionAngleWorkerProcess, WorkerProcessDataIterable, MPParams["ChunkSize"])
  448     elif re.match("^InMemory$", MPParams["InputDataMode"], re.I):
  449         Results = ProcessPool.map(TorsionAngleWorkerProcess, WorkerProcessDataIterable, MPParams["ChunkSize"])
  450     else:
  451         MiscUtil.PrintError(
  452             'The value, %s, specified for "--inputDataMode" is not supported.' % (MPParams["InputDataMode"])
  453         )
  454 
  455     TorsionMols = []
  456     TorsionEnergies = []
  457     TorsionAngles = []
  458 
  459     for Result in Results:
  460         EncodedTorsionMol, CalcStatus, Angle, Energy = Result
  461 
  462         if not CalcStatus:
  463             return (Mol, False, None, None, None)
  464 
  465         if EncodedTorsionMol is None:
  466             return (Mol, False, None, None, None)
  467         TorsionMol = RDKitUtil.MolFromBase64EncodedMolString(EncodedTorsionMol)
  468 
  469         if OptionsInfo["RemoveHydrogens"]:
  470             TorsionMol = Chem.RemoveHs(TorsionMol)
  471 
  472         TorsionMols.append(TorsionMol)
  473         TorsionEnergies.append(Energy)
  474         TorsionAngles.append(Angle)
  475 
  476     return (Mol, True, TorsionMols, TorsionEnergies, TorsionAngles)

  530 def GenerateBase64EncodedMolStringsWithTorsionScanInfo(
  531     Mol,
  532     MolIndex,
  533     TorsionMols,
  534     TorsionAngles,
  535     TorsionID,
  536     TorsionPattern,
  537     TorsionPatternMol,
  538     TorsionMatches,
  539     PropertyPickleFlags=Chem.PropertyPickleOptions.AllProps,
  540 ):
  541     """Set up an iterator for generating base64 encoded molecule string for
  542     a torsion in a molecule along with appropriate trosion scan information.
  543     """
  544 
  545     for Index, TorsionMol in enumerate(TorsionMols):
  546         yield (
  547             [MolIndex, None, None, TorsionAngles[Index], TorsionID, TorsionPattern, None, TorsionMatches]
  548             if (Mol is None or TorsionMol is None)
  549             else [
  550                 MolIndex,
  551                 RDKitUtil.MolToBase64EncodedMolString(Mol, PropertyPickleFlags),
  552                 RDKitUtil.MolToBase64EncodedMolString(TorsionMol, PropertyPickleFlags),
  553                 TorsionAngles[Index],
  554                 TorsionID,
  555                 TorsionPattern,
  556                 RDKitUtil.MolToBase64EncodedMolString(TorsionPatternMol, PropertyPickleFlags),
  557                 TorsionMatches,
  558             ]
  559         )

  562 def PerformMinimizationAndTorsionScan(Mol, MolNum, UseMultiProcessingAtTorsionAnglesLevel=False):
  563     """Perform minimization and torsions scan."""
  564 
  565     if not OptionsInfo["QuietMode"]:
  566         MiscUtil.PrintInfo("\nProcessing molecule %s..." % (RDKitUtil.GetMolName(Mol, MolNum)))
  567 
  568     Mol = AddHydrogens(Mol)
  569 
  570     if not OptionsInfo["Infile3D"]:
  571         Mol, MinimizationCalcStatus = MinimizeMolecule(Mol, MolNum)
  572         if not MinimizationCalcStatus:
  573             return (Mol, False, False, False)
  574 
  575     TorsionsMolInfo = SetupTorsionsMolInfo(Mol, MolNum)
  576     if TorsionsMolInfo["NumOfMatches"] == 0:
  577         return (Mol, True, False, False)
  578 
  579     Mol, ScanCalcStatus = ScanAllTorsionsInMol(Mol, TorsionsMolInfo, MolNum, UseMultiProcessingAtTorsionAnglesLevel)
  580     if not ScanCalcStatus:
  581         return (Mol, True, True, False)
  582 
  583     return (Mol, True, True, True)

  586 def ScanAllTorsionsInMol(Mol, TorsionsMolInfo, MolNum, UseMultiProcessingAtTorsionAnglesLevel=False):
  587     """Perform scans on all torsions in a molecule."""
  588 
  589     if TorsionsMolInfo["NumOfMatches"] == 0:
  590         return Mol, True
  591 
  592     MolName = RDKitUtil.GetMolName(Mol, MolNum)
  593 
  594     FirstTorsionMode = OptionsInfo["FirstTorsionMode"]
  595     TorsionsPatternsInfo = OptionsInfo["TorsionsPatternsInfo"]
  596 
  597     TorsionPatternCount, TorsionScanCount, TorsionMatchCount = [0] * 3

  599

  601         TorsionPatternCount += 1
  602         TorsionPattern = TorsionsPatternsInfo["Pattern"][TorsionID]

  604

  606

  608             continue
  609 
  610         if FirstTorsionMode and TorsionPatternCount > 1:
  611             if not OptionsInfo["QuietMode"]:
  612                 MiscUtil.PrintWarning(
  613                     'Already scaned first torsion pattern, "%s" for molecule %s during "%s" value of "--modeTorsions" option . Abandoning torsion scan...\n'
  614                     % (TorsionPattern, MolName, OptionsInfo["ModeTorsions"])
  615                 )
  616             break
  617 
  618         for Index, TorsionMatches in enumerate(TorsionsMatches):
  619             TorsionMatchNum = Index + 1
  620             TorsionMatchCount += 1
  621 
  622             if TorsionMatchCount > TorsionMaxMatches:
  623                 if not OptionsInfo["QuietMode"]:
  624                     MiscUtil.PrintWarning(
  625                         'Already scaned a maximum of %s torsion matches for molecule %s specified by "--torsionMaxMatches" option. Abandoning torsion scan...\n'
  626                         % (TorsionMaxMatches, MolName)
  627                     )
  628                 break
  629 
  630             TmpMol, TorsionScanStatus, TorsionMols, TorsionEnergies, TorsionAngles = ScanSingleTorsionInMol(
  631                 Mol,
  632                 TorsionID,
  633                 TorsionPattern,
  634                 TorsionPatternMol,
  635                 TorsionMatches,
  636                 TorsionMatchNum,
  637                 MolNum,
  638                 UseMultiProcessingAtTorsionAnglesLevel,
  639             )

  641                 continue
  642 
  643             TorsionScanCount += 1
  644             GenerateOutputFiles(Mol, MolNum, TorsionID, TorsionMatchNum, TorsionMols, TorsionEnergies, TorsionAngles)
  645 
  646         if TorsionMatchCount > TorsionMaxMatches:
  647             break
  648 
  649     if OptionsInfo["RemoveHydrogens"]:
  650         Mol = Chem.RemoveHs(Mol)
  651 
  652     if TorsionScanCount:
  653         GenerateStartingTorsionScanStructureOutfile(Mol, MolNum)
  654 
  655     Status = True if TorsionScanCount else False
  656 
  657     return (Mol, Status)

  772 def SetupTorsionsMolInfo(Mol, MolNum=None):
  773     """Setup torsions info for a molecule."""
  774 
  775     TorsionsPatternsInfo = OptionsInfo["TorsionsPatternsInfo"]
  776 
  777     # Initialize...
  778     TorsionsMolInfo = {}
  779     TorsionsMolInfo["IDs"] = []
  780     TorsionsMolInfo["NumOfMatches"] = 0
  781     TorsionsMolInfo["Matches"] = {}
  782     for TorsionID in TorsionsPatternsInfo["IDs"]:
  783         TorsionsMolInfo["IDs"].append(TorsionID)
  784         TorsionsMolInfo["Matches"][TorsionID] = None
  785 
  786     MolName = RDKitUtil.GetMolName(Mol, MolNum)
  787     UseChirality = OptionsInfo["UseChirality"]
  788 
  789     for TorsionID in TorsionsPatternsInfo["IDs"]:
  790         # Match torsions..
  791         TorsionPattern = TorsionsPatternsInfo["Pattern"][TorsionID]
  792         TorsionPatternMol = TorsionsPatternsInfo["Mol"][TorsionID]
  793         TorsionsMatches = RDKitUtil.FilterSubstructureMatchesByAtomMapNumbers(
  794             Mol, TorsionPatternMol, Mol.GetSubstructMatches(TorsionPatternMol, useChirality=UseChirality)
  795         )
  796 
  797         # Validate tosion matches...
  798         ValidTorsionsMatches = []
  799         for Index, TorsionMatch in enumerate(TorsionsMatches):
  800             if len(TorsionMatch) != 4:
  801                 if not OptionsInfo["QuietMode"]:
  802                     MiscUtil.PrintWarning(
  803                         "Ignoring invalid torsion match to atom indices, %s, for torsion pattern, %s, in molecule %s: It must match exactly 4 atoms."
  804                         % (TorsionMatch, TorsionPattern, MolName)
  805                     )
  806                 continue
  807 
  808             if not RDKitUtil.AreAtomIndicesSequentiallyConnected(Mol, TorsionMatch):
  809                 if not OptionsInfo["QuietMode"]:
  810                     MiscUtil.PrintInfo("")
  811                     MiscUtil.PrintWarning(
  812                         "Invalid torsion match to atom indices, %s, for torsion pattern, %s, in molecule %s: Matched atom indices must be sequentially connected."
  813                         % (TorsionMatch, TorsionPattern, MolName)
  814                     )
  815                     MiscUtil.PrintWarning("Reordering matched atom indices in a sequentially connected manner...")
  816 
  817                 Status, ReorderdTorsionMatch = RDKitUtil.ReorderAtomIndicesInSequentiallyConnectedManner(
  818                     Mol, TorsionMatch
  819                 )
  820                 if Status:
  821                     TorsionMatch = ReorderdTorsionMatch
  822                     if not OptionsInfo["QuietMode"]:
  823                         MiscUtil.PrintWarning(
  824                             "Successfully reordered torsion match to atom indices, %s, for torsion pattern, %s, in molecule %s: Matched atom indices are now sequentially connected."
  825                             % (TorsionMatch, TorsionPattern, MolName)
  826                         )
  827                 else:
  828                     if not OptionsInfo["QuietMode"]:
  829                         MiscUtil.PrintWarning(
  830                             "Ignoring torsion match. Failed to reorder torsion match to atom indices, %s, for torsion pattern, %s, in molecule %s: Matched atom indices are not sequentially connected."
  831                             % (TorsionMatch, TorsionPattern, MolName)
  832                         )
  833                     continue
  834 
  835             Bond = Mol.GetBondBetweenAtoms(TorsionMatch[1], TorsionMatch[2])
  836             if Bond.IsInRing():
  837                 if not OptionsInfo["QuietMode"]:
  838                     MiscUtil.PrintWarning(
  839                         "Ignoring invalid torsion match to atom indices, %s, for torsion pattern, %s, in molecule %s: Matched atom indices, %s and %s, are not allowed to be in a ring."
  840                         % (TorsionMatch, TorsionPattern, MolName, TorsionMatch[1], TorsionMatch[2])
  841                     )
  842                 continue
  843 
  844             # Filter matched torsions...
  845             if OptionsInfo["FilterTorsionsByAtomIndicesMode"]:
  846                 InvalidAtomIndices = []
  847                 for AtomIndex in TorsionMatch:
  848                     if AtomIndex not in OptionsInfo["TorsionsFilterByAtomIndicesList"]:
  849                         InvalidAtomIndices.append(AtomIndex)
  850                 if len(InvalidAtomIndices):
  851                     if not OptionsInfo["QuietMode"]:
  852                         MiscUtil.PrintWarning(
  853                             'Ignoring invalid torsion match to atom indices, %s, for torsion pattern, %s, in molecule %s: Matched atom indices, %s,  must be present in the list, %s, specified using option "--torsionsFilterbyAtomIndices".'
  854                             % (
  855                                 TorsionMatch,
  856                                 TorsionPattern,
  857                                 MolName,
  858                                 InvalidAtomIndices,
  859                                 OptionsInfo["TorsionsFilterByAtomIndicesList"],
  860                             )
  861                         )
  862                     continue
  863 
  864             ValidTorsionsMatches.append(TorsionMatch)
  865 
  866         # Track valid matches...
  867         if len(ValidTorsionsMatches):
  868             TorsionsMolInfo["NumOfMatches"] += len(ValidTorsionsMatches)
  869             TorsionsMolInfo["Matches"][TorsionID] = ValidTorsionsMatches
  870 
  871     if TorsionsMolInfo["NumOfMatches"] == 0:
  872         if not OptionsInfo["QuietMode"]:
  873             MiscUtil.PrintWarning("Failed to match any torsions  in molecule %s" % (MolName))
  874 
  875     return TorsionsMolInfo

  904 def MinimizeMolecule(Mol, MolNum=None):
  905     """Generate and minimize conformers for a molecule to get the lowest energy conformer."""
  906 
  907     if not OptionsInfo["QuietMode"]:
  908         MiscUtil.PrintInfo("Minimizing molecule %s..." % (RDKitUtil.GetMolName(Mol, MolNum)))
  909 
  910     ConfIDs = EmbedMolecule(Mol, MolNum)
  911     if not len(ConfIDs):
  912         if not OptionsInfo["QuietMode"]:
  913             MolName = RDKitUtil.GetMolName(Mol, MolNum)
  914             MiscUtil.PrintWarning("Minimization couldn't be performed for molecule %s: Embedding failed...\n" % MolName)
  915         return (Mol, False)
  916 
  917     CalcEnergyMap = {}
  918     for ConfID in ConfIDs:
  919         try:
  920             if OptionsInfo["UseUFF"]:
  921                 Status = AllChem.UFFOptimizeMolecule(Mol, confId=ConfID, maxIters=OptionsInfo["MaxIters"])
  922             elif OptionsInfo["UseMMFF"]:
  923                 Status = AllChem.MMFFOptimizeMolecule(
  924                     Mol, confId=ConfID, maxIters=OptionsInfo["MaxIters"], mmffVariant=OptionsInfo["MMFFVariant"]
  925                 )
  926             else:
  927                 MiscUtil.PrintError(
  928                     "Minimization couldn't be performed: Specified forcefield, %s, is not supported"
  929                     % OptionsInfo["ForceField"]
  930                 )
  931         except (ValueError, RuntimeError, Chem.rdchem.KekulizeException) as ErrMsg:
  932             if not OptionsInfo["QuietMode"]:
  933                 MolName = RDKitUtil.GetMolName(Mol, MolNum)
  934                 MiscUtil.PrintWarning("Minimization couldn't be performed for molecule %s:\n%s\n" % (MolName, ErrMsg))
  935             return (Mol, False)
  936 
  937         EnergyStatus, Energy = GetEnergy(Mol, ConfID)
  938         if not EnergyStatus:
  939             if not OptionsInfo["QuietMode"]:
  940                 MolName = RDKitUtil.GetMolName(Mol, MolNum)
  941                 MiscUtil.PrintWarning(
  942                     "Failed to retrieve calculated energy for conformation number %d of molecule %s. Try again after removing any salts or cleaing up the molecule...\n"
  943                     % (ConfID, MolName)
  944                 )
  945             return (Mol, False)
  946 
  947         if Status != 0:
  948             if not OptionsInfo["QuietMode"]:
  949                 MolName = RDKitUtil.GetMolName(Mol, MolNum)
  950                 MiscUtil.PrintWarning(
  951                     'Minimization failed to converge for conformation number %d of molecule %s in %d steps. Try using higher value for "--maxIters" option...\n'
  952                     % (ConfID, MolName, OptionsInfo["MaxIters"])
  953                 )
  954 
  955         CalcEnergyMap[ConfID] = Energy
  956 
  957     SortedConfIDs = sorted(ConfIDs, key=lambda ConfID: CalcEnergyMap[ConfID])
  958     MinEnergyConfID = SortedConfIDs[0]
  959 
  960     for ConfID in [Conf.GetId() for Conf in Mol.GetConformers()]:
  961         if ConfID == MinEnergyConfID:
  962             continue
  963         Mol.RemoveConformer(ConfID)
  964 
  965     # Set ConfID to 0 for MinEnergyConf...
  966     Mol.GetConformer(MinEnergyConfID).SetId(0)
  967 
  968     return (Mol, True)

  971 def ConstrainAndMinimizeMolecule(Mol, RefMolCore, RefMolMatches=None, MolNum=None):
  972     """Constrain and Minimize molecule."""
  973 
  974     # Setup forcefield function to use for constrained minimization...
  975     ForceFieldFunction = None
  976     ForceFieldName = None
  977     if OptionsInfo["UseUFF"]:
  978         ForceFieldFunction = lambda mol, confId=-1: AllChem.UFFGetMoleculeForceField(mol, confId=confId)
  979         ForceFieldName = "UFF"
  980     else:
  981         ForceFieldFunction = lambda mol, confId=-1: AllChem.MMFFGetMoleculeForceField(
  982             mol, AllChem.MMFFGetMoleculeProperties(mol, mmffVariant=OptionsInfo["MMFFVariant"]), confId=confId
  983         )
  984         ForceFieldName = "MMFF"
  985 
  986     if ForceFieldFunction is None:
  987         if not OptionsInfo["QuietMode"]:
  988             MiscUtil.PrintWarning(
  989                 "Failed to setup forcefield %s for molecule: %s\n" % (ForceFieldName, RDKitUtil.GetMolName(Mol, MolNum))
  990             )
  991         return (None, False, None)
  992 
  993     MaxConfs = OptionsInfo["MaxConfsTorsion"]
  994     EnforceChirality = OptionsInfo["EnforceChirality"]
  995     UseExpTorsionAnglePrefs = OptionsInfo["UseExpTorsionAnglePrefs"]
  996     ETVersion = OptionsInfo["ETVersion"]
  997     UseBasicKnowledge = OptionsInfo["UseBasicKnowledge"]
  998     UseTethers = OptionsInfo["UseTethers"]
  999 
 1000     CalcEnergyMap = {}
 1001     MolConfsMap = {}
 1002     ConfIDs = [ConfID for ConfID in range(0, MaxConfs)]
 1003 
 1004     for ConfID in ConfIDs:
 1005         try:
 1006             MolConf = Chem.Mol(Mol)
 1007             RDKitUtil.ConstrainAndEmbed(
 1008                 MolConf,
 1009                 RefMolCore,
 1010                 coreMatchesMol=RefMolMatches,
 1011                 useTethers=UseTethers,
 1012                 coreConfId=-1,
 1013                 randomseed=ConfID,
 1014                 getForceField=ForceFieldFunction,
 1015                 enforceChirality=EnforceChirality,
 1016                 useExpTorsionAnglePrefs=UseExpTorsionAnglePrefs,
 1017                 useBasicKnowledge=UseBasicKnowledge,
 1018                 ETversion=ETVersion,
 1019             )
 1020         except (ValueError, RuntimeError, Chem.rdchem.KekulizeException) as ErrMsg:
 1021             if not OptionsInfo["QuietMode"]:
 1022                 MolName = RDKitUtil.GetMolName(Mol, MolNum)
 1023                 MiscUtil.PrintWarning(
 1024                     "Constrained embedding couldn't  be performed for molecule %s:\n%s\n"
 1025                     % (RDKitUtil.GetMolName(Mol, MolNum), ErrMsg)
 1026                 )
 1027             return (None, False, None)
 1028 
 1029         EnergyStatus, Energy = GetEnergy(MolConf)
 1030 
 1031         if not EnergyStatus:
 1032             if not OptionsInfo["QuietMode"]:
 1033                 MolName = RDKitUtil.GetMolName(Mol, MolNum)
 1034                 MiscUtil.PrintWarning(
 1035                     "Failed to retrieve calculated energy for conformation number %d of molecule %s. Try again after removing any salts or cleaing up the molecule...\n"
 1036                     % (ConfID, MolName)
 1037                 )
 1038             return (None, False, None)
 1039 
 1040         CalcEnergyMap[ConfID] = Energy
 1041         MolConfsMap[ConfID] = MolConf
 1042 
 1043     SortedConfIDs = sorted(ConfIDs, key=lambda ConfID: CalcEnergyMap[ConfID])
 1044     MinEnergyConfID = SortedConfIDs[0]
 1045 
 1046     MinEnergy = CalcEnergyMap[MinEnergyConfID]
 1047     MinEnergyMolConf = MolConfsMap[MinEnergyConfID]
 1048 
 1049     MinEnergyMolConf.ClearProp("EmbedRMS")
 1050 
 1051     return (MinEnergyMolConf, True, MinEnergy)
