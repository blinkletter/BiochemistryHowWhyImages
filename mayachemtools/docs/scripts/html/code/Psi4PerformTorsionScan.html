  143 def ProcessMoleculesUsingSingleProcess(Mols):
  144     """Process molecules to perform torsion scan using a single process."""
  145 
  146     # Intialize Psi4...
  147     MiscUtil.PrintInfo("\nInitializing Psi4...")
  148     Psi4Handle = Psi4Util.InitializePsi4(
  149         Psi4RunParams=OptionsInfo["Psi4RunParams"],
  150         Psi4OptionsParams=OptionsInfo["Psi4OptionsParams"],
  151         PrintVersion=True,
  152         PrintHeader=True,
  153     )
  154     OptionsInfo["psi4"] = Psi4Handle
  155 
  156     # Setup max iterations global variable...
  157     Psi4Util.UpdatePsi4OptionsParameters(Psi4Handle, {"GEOM_MAXITER": OptionsInfo["MaxIters"]})
  158 
  159     # Setup conversion factor for energy units...
  160     SetupEnergyConversionFactor(Psi4Handle)
  161 
  162     MolInfoText = "first molecule"
  163     if not OptionsInfo["FirstMolMode"]:
  164         MolInfoText = "all molecules"
  165 
  166     if OptionsInfo["TorsionMinimize"]:
  167         MiscUtil.PrintInfo(
  168             "\nPerforming torsion scan on %s by generating conformation ensembles for specific torsion angles and constrained energy minimization of the ensembles..."
  169             % (MolInfoText)
  170         )
  171     else:
  172         MiscUtil.PrintInfo(
  173             "\nPerforming torsion scan on %s by skipping generation of conformation ensembles for specific torsion angles and constrained energy minimization of the ensembles..."
  174             % (MolInfoText)
  175         )
  176 
  177     SetupTorsionsPatternsInfo()
  178 
  179     (MolCount, ValidMolCount, TorsionsMissingCount, MinimizationFailedCount, TorsionsScanFailedCount) = [0] * 5
  180 
  181     for Mol in Mols:
  182         MolCount += 1
  183 
  184         if OptionsInfo["FirstMolMode"] and MolCount > 1:
  185             MolCount -= 1
  186             break
  187 
  188         if not CheckAndValidateMolecule(Mol, MolCount):
  189             continue
  190 
  191         # Setup 2D coordinates for SMILES input file...
  192         if OptionsInfo["SMILESInfileStatus"]:
  193             AllChem.Compute2DCoords(Mol)
  194 
  195         ValidMolCount += 1
  196 
  197         Mol, MinimizationCalcStatus, TorsionsMatchStatus, TorsionsScanCalcStatus = PerformMinimizationAndTorsionScan(
  198             Mol, MolCount
  199         )
  200 
  201         if not MinimizationCalcStatus:
  202             MinimizationFailedCount += 1
  203             continue
  204 
  205         if not TorsionsMatchStatus:
  206             TorsionsMissingCount += 1
  207             continue
  208 
  209         if not TorsionsScanCalcStatus:
  210             TorsionsScanFailedCount += 1
  211             continue
  212 
  213     return (MolCount, ValidMolCount, MinimizationFailedCount, TorsionsMissingCount, TorsionsScanFailedCount)

  227 def ProcessMoleculesUsingMultipleProcessesAtMoleculesLevel(Mols):
  228     """Process molecules to perform torsion scan using multiprocessing at molecules level."""
  229 
  230     MolInfoText = "first molecule"
  231     if not OptionsInfo["FirstMolMode"]:
  232         MolInfoText = "all molecules"
  233 
  234     if OptionsInfo["TorsionMinimize"]:
  235         MiscUtil.PrintInfo(
  236             "\nPerforming torsion scan on %s using multiprocessing at molecules level by generating conformation ensembles for specific torsion angles and constrained energy minimization of the ensembles..."
  237             % (MolInfoText)
  238         )
  239     else:
  240         MiscUtil.PrintInfo(
  241             "\nPerforming torsion scan %s using multiprocessing at molecules level by skipping generation of conformation ensembles for specific torsion angles and constrained energy minimization of the ensembles..."
  242             % (MolInfoText)
  243         )
  244 
  245     MPParams = OptionsInfo["MPParams"]
  246 
  247     # Setup data for initializing a worker process...
  248     InitializeWorkerProcessArgs = (
  249         MiscUtil.ObjectToBase64EncodedString(Options),
  250         MiscUtil.ObjectToBase64EncodedString(OptionsInfo),
  251     )
  252 
  253     if OptionsInfo["FirstMolMode"]:
  254         Mol = Mols[0]
  255         Mols = [Mol]
  256 
  257     # Setup a encoded mols data iterable for a worker process...
  258     WorkerProcessDataIterable = RDKitUtil.GenerateBase64EncodedMolStrings(Mols)
  259 
  260     # Setup process pool along with data initialization for each process...
  261     MiscUtil.PrintInfo(
  262         "\nConfiguring multiprocessing using %s method..."
  263         % ("mp.Pool.imap()" if re.match("^Lazy$", MPParams["InputDataMode"], re.I) else "mp.Pool.map()")
  264     )
  265     MiscUtil.PrintInfo(
  266         "NumProcesses: %s; InputDataMode: %s; ChunkSize: %s\n"
  267         % (
  268             MPParams["NumProcesses"],
  269             MPParams["InputDataMode"],
  270             ("automatic" if MPParams["ChunkSize"] is None else MPParams["ChunkSize"]),
  271         )
  272     )
  273 
  274     ProcessPool = mp.Pool(MPParams["NumProcesses"], InitializeWorkerProcess, InitializeWorkerProcessArgs)
  275 
  276     # Start processing...
  277     if re.match("^Lazy$", MPParams["InputDataMode"], re.I):
  278         Results = ProcessPool.imap(WorkerProcess, WorkerProcessDataIterable, MPParams["ChunkSize"])
  279     elif re.match("^InMemory$", MPParams["InputDataMode"], re.I):
  280         Results = ProcessPool.map(WorkerProcess, WorkerProcessDataIterable, MPParams["ChunkSize"])
  281     else:
  282         MiscUtil.PrintError(
  283             'The value, %s, specified for "--inputDataMode" is not supported.' % (MPParams["InputDataMode"])
  284         )
  285 
  286     (MolCount, ValidMolCount, TorsionsMissingCount, MinimizationFailedCount, TorsionsScanFailedCount) = [0] * 5
  287 
  288     for Result in Results:
  289         MolCount += 1
  290 
  291         MolIndex, EncodedMol, MinimizationCalcStatus, TorsionsMatchStatus, TorsionsScanCalcStatus = Result
  292 
  293         if EncodedMol is None:
  294             continue
  295         ValidMolCount += 1
  296 
  297         if not MinimizationCalcStatus:
  298             MinimizationFailedCount += 1
  299             continue
  300 
  301         if not TorsionsMatchStatus:
  302             TorsionsMissingCount += 1
  303             continue
  304 
  305         if not TorsionsScanCalcStatus:
  306             TorsionsScanFailedCount += 1
  307             continue
  308 
  309         Mol = RDKitUtil.MolFromBase64EncodedMolString(EncodedMol)
  310 
  311     return (MolCount, ValidMolCount, MinimizationFailedCount, TorsionsMissingCount, TorsionsScanFailedCount)

  334 def WorkerProcess(EncodedMolInfo):
  335     """Process data for a worker process."""
  336 
  337     if not OptionsInfo["Psi4Initialized"]:
  338         InitializePsi4ForWorkerProcess()
  339 
  340     MolIndex, EncodedMol = EncodedMolInfo
  341 
  342     MolNum = MolIndex + 1
  343     (MinimizationCalcStatus, TorsionsMatchStatus, TorsionsScanCalcStatus) = [False] * 3
  344 
  345     if EncodedMol is None:
  346         return [MolIndex, None, MinimizationCalcStatus, TorsionsMatchStatus, TorsionsScanCalcStatus]
  347 
  348     Mol = RDKitUtil.MolFromBase64EncodedMolString(EncodedMol)
  349     if not CheckAndValidateMolecule(Mol, MolNum):
  350         return [MolIndex, None, MinimizationCalcStatus, TorsionsMatchStatus, TorsionsScanCalcStatus]
  351 
  352     # Setup 2D coordinates for SMILES input file...
  353     if OptionsInfo["SMILESInfileStatus"]:
  354         AllChem.Compute2DCoords(Mol)
  355 
  356     Mol, MinimizationCalcStatus, TorsionsMatchStatus, TorsionsScanCalcStatus = PerformMinimizationAndTorsionScan(
  357         Mol, MolNum
  358     )
  359 
  360     return [
  361         MolIndex,
  362         RDKitUtil.MolToBase64EncodedMolString(
  363             Mol, PropertyPickleFlags=Chem.PropertyPickleOptions.MolProps | Chem.PropertyPickleOptions.PrivateProps
  364         ),
  365         MinimizationCalcStatus,
  366         TorsionsMatchStatus,
  367         TorsionsScanCalcStatus,
  368     ]

  428 def ScanSingleTorsionInMolUsingMultipleProcessesAtTorsionAnglesLevel(
  429     Mol, TorsionID, TorsionPattern, TorsionPatternMol, TorsionMatches, MolNum
  430 ):
  431     """Perform torsion scan for a molecule using multiple processses at torsion angles
  432     level along with constrained energy minimization.
  433     """
  434 
  435     if OptionsInfo["MPLevelMoleculesMode"]:
  436         MiscUtil.PrintError(
  437             "Single torison scanning for a molecule is not allowed in multiprocessing mode at molecules level.\n"
  438         )
  439 
  440     Mols, Angles = SetupMolsForSingleTorsionScanInMol(Mol, TorsionMatches, MolNum)
  441 
  442     MPParams = OptionsInfo["MPParams"]
  443 
  444     # Setup data for initializing a worker process...
  445 
  446     # Track and avoid encoding TorsionsPatternsInfo as it contains RDKit molecule object...
  447     TorsionsPatternsInfo = OptionsInfo["TorsionsPatternsInfo"]
  448     OptionsInfo["TorsionsPatternsInfo"] = None
  449 
  450     InitializeWorkerProcessArgs = (
  451         MiscUtil.ObjectToBase64EncodedString(Options),
  452         MiscUtil.ObjectToBase64EncodedString(OptionsInfo),
  453     )
  454 
  455     # Restore TorsionsPatternsInfo...
  456     OptionsInfo["TorsionsPatternsInfo"] = TorsionsPatternsInfo
  457 
  458     # Setup a encoded mols data iterable for a worker process...
  459     WorkerProcessDataIterable = GenerateBase64EncodedMolStringsWithTorsionScanInfo(
  460         Mol, (MolNum - 1), Mols, Angles, TorsionID, TorsionPattern, TorsionPatternMol, TorsionMatches
  461     )
  462 
  463     # Setup process pool along with data initialization for each process...
  464     MiscUtil.PrintInfo(
  465         "\nConfiguring multiprocessing using %s method..."
  466         % ("mp.Pool.imap()" if re.match("^Lazy$", MPParams["InputDataMode"], re.I) else "mp.Pool.map()")
  467     )
  468     MiscUtil.PrintInfo(
  469         "NumProcesses: %s; InputDataMode: %s; ChunkSize: %s\n"
  470         % (
  471             MPParams["NumProcesses"],
  472             MPParams["InputDataMode"],
  473             ("automatic" if MPParams["ChunkSize"] is None else MPParams["ChunkSize"]),
  474         )
  475     )
  476 
  477     ProcessPool = mp.Pool(MPParams["NumProcesses"], InitializeTorsionAngleWorkerProcess, InitializeWorkerProcessArgs)
  478 
  479     # Start processing...
  480     if re.match("^Lazy$", MPParams["InputDataMode"], re.I):
  481         Results = ProcessPool.imap(TorsionAngleWorkerProcess, WorkerProcessDataIterable, MPParams["ChunkSize"])
  482     elif re.match("^InMemory$", MPParams["InputDataMode"], re.I):
  483         Results = ProcessPool.map(TorsionAngleWorkerProcess, WorkerProcessDataIterable, MPParams["ChunkSize"])
  484     else:
  485         MiscUtil.PrintError(
  486             'The value, %s, specified for "--inputDataMode" is not supported.' % (MPParams["InputDataMode"])
  487         )
  488 
  489     TorsionMols = []
  490     TorsionEnergies = []
  491     TorsionAngles = []
  492 
  493     for Result in Results:
  494         EncodedTorsionMol, CalcStatus, Angle, Energy = Result
  495 
  496         if not CalcStatus:
  497             return (Mol, False, None, None, None)
  498 
  499         if EncodedTorsionMol is None:
  500             return (Mol, False, None, None, None)
  501         TorsionMol = RDKitUtil.MolFromBase64EncodedMolString(EncodedTorsionMol)
  502 
  503         TorsionMols.append(TorsionMol)
  504         TorsionEnergies.append(Energy)
  505         TorsionAngles.append(Angle)
  506 
  507     return (Mol, True, TorsionMols, TorsionEnergies, TorsionAngles)

  569 def GenerateBase64EncodedMolStringsWithTorsionScanInfo(
  570     Mol,
  571     MolIndex,
  572     TorsionMols,
  573     TorsionAngles,
  574     TorsionID,
  575     TorsionPattern,
  576     TorsionPatternMol,
  577     TorsionMatches,
  578     PropertyPickleFlags=Chem.PropertyPickleOptions.AllProps,
  579 ):
  580     """Set up an iterator for generating base64 encoded molecule string for
  581     a torsion in a molecule along with appropriate trosion scan information.
  582     """
  583 
  584     for Index, TorsionMol in enumerate(TorsionMols):
  585         yield (
  586             [MolIndex, None, None, TorsionAngles[Index], TorsionID, TorsionPattern, None, TorsionMatches]
  587             if (Mol is None or TorsionMol is None)
  588             else [
  589                 MolIndex,
  590                 RDKitUtil.MolToBase64EncodedMolString(Mol, PropertyPickleFlags),
  591                 RDKitUtil.MolToBase64EncodedMolString(TorsionMol, PropertyPickleFlags),
  592                 TorsionAngles[Index],
  593                 TorsionID,
  594                 TorsionPattern,
  595                 RDKitUtil.MolToBase64EncodedMolString(TorsionPatternMol, PropertyPickleFlags),
  596                 TorsionMatches,
  597             ]
  598         )

  857 def SetupTorsionsMolInfo(Mol, MolNum=None):
  858     """Setup torsions info for a molecule."""
  859 
  860     TorsionsPatternsInfo = OptionsInfo["TorsionsPatternsInfo"]
  861 
  862     # Initialize...
  863     TorsionsMolInfo = {}
  864     TorsionsMolInfo["IDs"] = []
  865     TorsionsMolInfo["NumOfMatches"] = 0
  866     TorsionsMolInfo["Matches"] = {}
  867     for TorsionID in TorsionsPatternsInfo["IDs"]:
  868         TorsionsMolInfo["IDs"].append(TorsionID)
  869         TorsionsMolInfo["Matches"][TorsionID] = None
  870 
  871     MolName = RDKitUtil.GetMolName(Mol, MolNum)
  872     UseChirality = OptionsInfo["UseChirality"]
  873 
  874     for TorsionID in TorsionsPatternsInfo["IDs"]:
  875         # Match torsions..
  876         TorsionPattern = TorsionsPatternsInfo["Pattern"][TorsionID]
  877         TorsionPatternMol = TorsionsPatternsInfo["Mol"][TorsionID]
  878         TorsionsMatches = RDKitUtil.FilterSubstructureMatchesByAtomMapNumbers(
  879             Mol, TorsionPatternMol, Mol.GetSubstructMatches(TorsionPatternMol, useChirality=UseChirality)
  880         )
  881 
  882         # Validate tosion matches...
  883         ValidTorsionsMatches = []
  884         for Index, TorsionMatch in enumerate(TorsionsMatches):
  885             if len(TorsionMatch) != 4:
  886                 if not OptionsInfo["QuietMode"]:
  887                     MiscUtil.PrintWarning(
  888                         "Ignoring invalid torsion match to atom indices, %s, for torsion pattern, %s, in molecule %s: It must match exactly 4 atoms."
  889                         % (TorsionMatch, TorsionPattern, MolName)
  890                     )
  891                 continue
  892 
  893             if not RDKitUtil.AreAtomIndicesSequentiallyConnected(Mol, TorsionMatch):
  894                 if not OptionsInfo["QuietMode"]:
  895                     MiscUtil.PrintInfo("")
  896                     MiscUtil.PrintWarning(
  897                         "Invalid torsion match to atom indices, %s, for torsion pattern, %s, in molecule %s: Matched atom indices must be sequentially connected."
  898                         % (TorsionMatch, TorsionPattern, MolName)
  899                     )
  900                     MiscUtil.PrintWarning("Reordering matched atom indices in a sequentially connected manner...")
  901 
  902                 Status, ReorderdTorsionMatch = RDKitUtil.ReorderAtomIndicesInSequentiallyConnectedManner(
  903                     Mol, TorsionMatch
  904                 )
  905                 if Status:
  906                     TorsionMatch = ReorderdTorsionMatch
  907                     if not OptionsInfo["QuietMode"]:
  908                         MiscUtil.PrintWarning(
  909                             "Successfully reordered torsion match to atom indices, %s, for torsion pattern, %s, in molecule %s: Matched atom indices are now sequentially connected."
  910                             % (TorsionMatch, TorsionPattern, MolName)
  911                         )
  912                 else:
  913                     if not OptionsInfo["QuietMode"]:
  914                         MiscUtil.PrintWarning(
  915                             "Ignoring torsion match. Failed to reorder torsion match to atom indices, %s, for torsion pattern, %s, in molecule %s: Matched atom indices are not sequentially connected."
  916                             % (TorsionMatch, TorsionPattern, MolName)
  917                         )
  918                     continue
  919 
  920             Bond = Mol.GetBondBetweenAtoms(TorsionMatch[1], TorsionMatch[2])
  921             if Bond.IsInRing():
  922                 if not OptionsInfo["QuietMode"]:
  923                     MiscUtil.PrintWarning(
  924                         "Ignoring invalid torsion match to atom indices, %s, for torsion pattern, %s, in molecule %s: Matched atom indices, %s and %s, are not allowed to be in a ring."
  925                         % (TorsionMatch, TorsionPattern, MolName, TorsionMatch[1], TorsionMatch[2])
  926                     )
  927                 continue
  928 
  929             # Filter matched torsions...
  930             if OptionsInfo["FilterTorsionsByAtomIndicesMode"]:
  931                 InvalidAtomIndices = []
  932                 for AtomIndex in TorsionMatch:
  933                     if AtomIndex not in OptionsInfo["TorsionsFilterByAtomIndicesList"]:
  934                         InvalidAtomIndices.append(AtomIndex)
  935                 if len(InvalidAtomIndices):
  936                     if not OptionsInfo["QuietMode"]:
  937                         MiscUtil.PrintWarning(
  938                             'Ignoring invalid torsion match to atom indices, %s, for torsion pattern, %s, in molecule %s: Matched atom indices, %s,  must be present in the list, %s, specified using option "--torsionsFilterbyAtomIndices".'
  939                             % (
  940                                 TorsionMatch,
  941                                 TorsionPattern,
  942                                 MolName,
  943                                 InvalidAtomIndices,
  944                                 OptionsInfo["TorsionsFilterByAtomIndicesList"],
  945                             )
  946                         )
  947                     continue
  948 
  949             ValidTorsionsMatches.append(TorsionMatch)
  950 
  951         # Track valid matches...
  952         if len(ValidTorsionsMatches):
  953             TorsionsMolInfo["NumOfMatches"] += len(ValidTorsionsMatches)
  954             TorsionsMolInfo["Matches"][TorsionID] = ValidTorsionsMatches
  955 
  956     if TorsionsMolInfo["NumOfMatches"] == 0:
  957         if not OptionsInfo["QuietMode"]:
  958             MiscUtil.PrintWarning("Failed to match any torsions  in molecule %s" % (MolName))
  959 
  960     return TorsionsMolInfo

  995 def GenerateAndMinimizeConformersUsingForceField(Mol, MolNum=None):

 1004         if not OptionsInfo["QuietMode"]:

 1010             "Performing initial minimization of molecule, %s, using forcefield by generating a conformation ensemble and selecting the lowest energy conformer - EmbedRMSDCutoff: %s; Size: %s; Size after RMSD filtering: %s"
 1011             % (
 1012                 MolName,
 1013                 OptionsInfo["ConfGenerationParams"]["EmbedRMSDCutoff"],
 1014                 OptionsInfo["ConfGenerationParams"]["MaxConfs"],
 1015                 len(ConfIDs),
 1016             )
 1017         )

 1020     CalcEnergyMap = {}

 1022         # Perform forcefield minimization...
 1023         Status, ConvergeStatus = MinimizeMoleculeUsingForceField(Mol, MolNum, ConfID)

 1039                 MiscUtil.PrintWarning(

 1222 def FilterConstrainedConformationsByRMSD(Mol, MolConfs, MolNum=None):
 1223     """Filter contarained conformations by RMSD."""
 1224 
 1225     EmbedRMSDCutoff = OptionsInfo["ConfGenerationParams"]["EmbedRMSDCutoff"]

 1231             )

 1237         if FirstMolConf:
 1238             FirstMolConf = False
 1239             SelectedMolConfs.append(MolConf)

 1243         ProbeMolConf = Chem.RemoveHs(Chem.Mol(MolConf))

 1250                 IgnoreConf = True
 1251                 break

 1255

 1257

 1261             % (RDKitUtil.GetMolName(Mol, MolNum), EmbedRMSDCutoff, len(MolConfs), len(SelectedMolConfs))
 1262         )

 1758             )

 1761             MiscUtil.PrintError(
 1762                 'The stop value, %d, specified for option "--torsionRange" in string "%s" must be  <= 360 to use scan range from 0 to 360.'
 1763                 % (TorsionStop, Options["--torsionRange"])
 1764             )
